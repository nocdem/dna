cmake_minimum_required(VERSION 3.16)
project(qgp_ckb_sphincs C)
if(NOT TARGET qgp_slhdsa)
 add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../slhdsa ${CMAKE_CURRENT_BINARY_DIR}/slhdsa)
endif()
add_library(qgp_ckb_sphincs STATIC qgp_ckb_sphincs.c)
target_include_directories(qgp_ckb_sphincs PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(qgp_ckb_sphincs PUBLIC qgp_slhdsa)
set_target_properties(qgp_ckb_sphincs PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
