cmake_minimum_required(VERSION 3.16)
project(qgp_slhdsa C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(qgp_slhdsa STATIC qgp_slhdsa.c)
target_include_directories(qgp_slhdsa PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set(common address.c fors.c merkle.c sign.c utils.c utilsx1.c wots.c wotsx1.c)
foreach(hash sha2 shake)
  foreach(level 128f 128s 192f 192s 256f 256s)
    set(name ${hash}_${level})
    set(src variant.c)
    foreach(file IN LISTS common)
      list(APPEND src upstream/${file})
    endforeach()
    list(APPEND src upstream/hash_${hash}.c upstream/thash_${hash}_simple.c)
    if(hash STREQUAL "sha2")
      list(APPEND src upstream/sha2.c)
    else()
      list(APPEND src upstream/fips202.c)
    endif()
    add_library(qgp_slhdsa_${name} OBJECT ${src})
    target_include_directories(qgp_slhdsa_${name} PRIVATE upstream)
    target_compile_definitions(qgp_slhdsa_${name} PRIVATE PARAMS=sphincs-${hash}-${level} QGP_VARIANT=qgp_spx_${name}_)
    if(MSVC)
      target_compile_options(qgp_slhdsa_${name} PRIVATE /FI${CMAKE_CURRENT_SOURCE_DIR}/namespace.h)
    else()
      target_compile_options(qgp_slhdsa_${name} PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/namespace.h)
    endif()
    set_target_properties(qgp_slhdsa_${name} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_sources(qgp_slhdsa PRIVATE $<TARGET_OBJECTS:qgp_slhdsa_${name}>)
  endforeach()
endforeach()
set_target_properties(qgp_slhdsa PROPERTIES POSITION_INDEPENDENT_CODE ON)
