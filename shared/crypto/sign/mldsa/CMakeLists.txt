cmake_minimum_required(VERSION 3.16)
project(qgp_mldsa LANGUAGES C)
get_filename_component(QGP_MLDSA_SHARED "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(QGP_MLDSA_DSA "${QGP_MLDSA_SHARED}/crypto/sign/dsa")
set(QGP_MLDSA_CORE)
foreach(name ntt packing poly polyvec reduce rounding sign symmetric-shake)
    list(APPEND QGP_MLDSA_CORE "${QGP_MLDSA_DSA}/${name}.c")
endforeach()
foreach(mode 2 3)
    add_library(qgp_mldsa_mode${mode} OBJECT ${QGP_MLDSA_CORE} mode_adapter.c)
    target_compile_features(qgp_mldsa_mode${mode} PRIVATE c_std_99)
    target_include_directories(qgp_mldsa_mode${mode} PRIVATE "${QGP_MLDSA_SHARED}")
    target_compile_definitions(qgp_mldsa_mode${mode} PRIVATE QGP_MLDSA_MODE=${mode})
    if(MSVC)
        target_compile_options(qgp_mldsa_mode${mode} PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/force_config.h")
    else()
        target_compile_options(qgp_mldsa_mode${mode} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/force_config.h")
    endif()
    set_target_properties(qgp_mldsa_mode${mode} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
add_library(qgp_mldsa STATIC qgp_mldsa.c entropy_unavailable.c
    $<TARGET_OBJECTS:qgp_mldsa_mode2> $<TARGET_OBJECTS:qgp_mldsa_mode3>)
if(TARGET dsa)
    target_link_libraries(qgp_mldsa PRIVATE dsa)
else()
    target_sources(qgp_mldsa PRIVATE "${QGP_MLDSA_DSA}/fips202.c")
endif()
target_compile_features(qgp_mldsa PUBLIC c_std_99)
target_include_directories(qgp_mldsa PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(qgp_mldsa PRIVATE "${QGP_MLDSA_SHARED}")
set_target_properties(qgp_mldsa PROPERTIES POSITION_INDEPENDENT_CODE ON)
