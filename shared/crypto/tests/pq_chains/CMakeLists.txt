cmake_minimum_required(VERSION 3.16)
project(pq_chain_conformance C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(PQ_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Werror)
endif()
if(PQ_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
get_filename_component(CRYPTO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${CRYPTO}/sign/dsa" existing_dsa)
add_subdirectory("${CRYPTO}/sign/algorand_falcon" algorand)
add_subdirectory("${CRYPTO}/sign/tidecoin_falcon" tidecoin)
add_subdirectory("${CRYPTO}/sign/ckb_sphincs" ckb)
add_subdirectory("${CRYPTO}/sign/mldsa" mldsa)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME pq_provenance COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_provenance.py")
add_executable(test_pq_chains test_pq_chains.c existing_dsa_probe.c test_boundaries.c)
target_link_libraries(test_pq_chains PRIVATE qgp_algorand_falcon qgp_tidecoin_falcon qgp_ckb_sphincs qgp_mldsa dsa)
set(FIXTURES algorand tidecoin tidecoin-historical slhdsa-nist ckb mldsa-nist existing-dsa87
  soqucoin-mldsa qtc-mldsa qtc-slhdsa qtc-nist circl-1.6.3-mldsa circl-1.6.5-mldsa
  circl-1.6.5-slhdsa asentum-noble redteam-algorand redteam-tidecoin redteam-slhdsa redteam-mldsa ckb-hash)
foreach(name IN LISTS FIXTURES)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/unpack_fixture.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/${name}.bin.gz" "${CMAKE_CURRENT_BINARY_DIR}/${name}.bin"
    RESULT_VARIABLE unpack_result)
  if(NOT unpack_result EQUAL 0)
    message(FATAL_ERROR "Fixture unavailable or invalid: ${name}")
  endif()
  if(EMSCRIPTEN)
    target_link_options(test_pq_chains PRIVATE "SHELL:--embed-file \"${CMAKE_CURRENT_BINARY_DIR}/${name}.bin@/${name}.bin\"")
    add_test(NAME "pq_${name}" COMMAND node "$<TARGET_FILE:test_pq_chains>" "/${name}.bin")
  else()
    add_test(NAME "pq_${name}" COMMAND test_pq_chains "${CMAKE_CURRENT_BINARY_DIR}/${name}.bin")
  endif()
endforeach()
if(EMSCRIPTEN)
  target_link_options(test_pq_chains PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sSTACK_SIZE=1048576
    -sINITIAL_MEMORY=67108864 -sALLOW_MEMORY_GROWTH=1 -sSINGLE_FILE=1)
endif()
add_executable(test_pq_entropy test_entropy.c)
target_link_libraries(test_pq_entropy PRIVATE qgp_tidecoin_falcon)
if(EMSCRIPTEN)
  target_link_options(test_pq_entropy PRIVATE "SHELL:--embed-file \"${CMAKE_CURRENT_BINARY_DIR}/tidecoin.bin@/tidecoin.bin\""
    -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sSTACK_SIZE=1048576 -sSINGLE_FILE=1)
  add_test(NAME pq_entropy COMMAND node "$<TARGET_FILE:test_pq_entropy>" /tidecoin.bin)
else()
  find_package(Threads REQUIRED)
  target_compile_definitions(test_pq_entropy PRIVATE PQ_NATIVE_THREADS=1)
  target_link_libraries(test_pq_entropy PRIVATE Threads::Threads)
  add_test(NAME pq_entropy COMMAND test_pq_entropy "${CMAKE_CURRENT_BINARY_DIR}/tidecoin.bin")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(test_pq_tidecoin_alloc test_tidecoin_alloc.c)
    target_link_libraries(test_pq_tidecoin_alloc PRIVATE qgp_tidecoin_falcon)
    target_link_options(test_pq_tidecoin_alloc PRIVATE -Wl,--wrap=malloc)
    add_test(NAME pq_tidecoin_alloc COMMAND test_pq_tidecoin_alloc "${CMAKE_CURRENT_BINARY_DIR}/tidecoin.bin")
    add_executable(test_pq_alloc_failures test_alloc_failures.c)
    target_link_libraries(test_pq_alloc_failures PRIVATE qgp_ckb_sphincs)
    target_link_options(test_pq_alloc_failures PRIVATE -Wl,--wrap=malloc)
    add_test(NAME pq_alloc_failures COMMAND test_pq_alloc_failures "${CMAKE_CURRENT_BINARY_DIR}/circl-1.6.5-slhdsa.bin")
  endif()
endif()
