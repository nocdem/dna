cmake_minimum_required(VERSION 3.16)
project(mldsa65_sign LANGUAGES C)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Configure with emcmake")
endif()
get_filename_component(DNA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${DNA_ROOT}/shared/crypto/sign/mldsa" "${CMAKE_CURRENT_BINARY_DIR}/mldsa")
add_executable(mldsa65_sign "${CMAKE_CURRENT_SOURCE_DIR}/../mldsa65-sign.c")
target_include_directories(mldsa65_sign PRIVATE "${DNA_ROOT}/shared")
target_link_libraries(mldsa65_sign PRIVATE qgp_mldsa)
set_target_properties(mldsa65_sign PROPERTIES OUTPUT_NAME mldsa65-sign SUFFIX .wasm)
target_link_options(mldsa65_sign PRIVATE -O2 --no-entry "-sSTANDALONE_WASM=1" "-sFILESYSTEM=0" "-sSTACK_SIZE=1048576" "-sINITIAL_MEMORY=4194304" "-sALLOW_MEMORY_GROWTH=0" "-sEXPORTED_FUNCTIONS=['_mldsa65_seed','_mldsa65_message','_mldsa65_public_key','_mldsa65_signature','_mldsa65_sign','_mldsa65_verify','_mldsa65_clear']" -Wl,--export-memory -Wl,--gc-sections -Wl,--strip-all)
