cmake_minimum_required(VERSION 3.16)
project(mldsa44_sign LANGUAGES C)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Configure with emcmake")
endif()
get_filename_component(DNA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${DNA_ROOT}/shared/crypto/sign/mldsa" "${CMAKE_CURRENT_BINARY_DIR}/mldsa")
add_executable(mldsa44_sign "${CMAKE_CURRENT_SOURCE_DIR}/../mldsa44-sign.c")
target_include_directories(mldsa44_sign PRIVATE "${DNA_ROOT}/shared")
target_link_libraries(mldsa44_sign PRIVATE qgp_mldsa)
set_target_properties(mldsa44_sign PROPERTIES OUTPUT_NAME mldsa44-sign SUFFIX .wasm)
target_link_options(mldsa44_sign PRIVATE -O2 --no-entry "-sSTANDALONE_WASM=1" "-sFILESYSTEM=0" "-sSTACK_SIZE=1048576" "-sINITIAL_MEMORY=4194304" "-sALLOW_MEMORY_GROWTH=0" "-sEXPORTED_FUNCTIONS=['_mldsa44_seed','_mldsa44_message','_mldsa44_public_key','_mldsa44_signature','_mldsa44_sign','_mldsa44_clear','_mldsa44_random','_mldsa44_verify']" -Wl,--export-memory -Wl,--gc-sections -Wl,--strip-all)
