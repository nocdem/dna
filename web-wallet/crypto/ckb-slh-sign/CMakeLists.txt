cmake_minimum_required(VERSION 3.16)
project(ckb_slh_sign LANGUAGES C)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Configure with emcmake")
endif()
get_filename_component(DNA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${DNA_ROOT}/shared/crypto/sign/slhdsa" "${CMAKE_CURRENT_BINARY_DIR}/slhdsa")
add_executable(ckb_slh_sign "${CMAKE_CURRENT_SOURCE_DIR}/../ckb-slh-sign.c")
target_include_directories(ckb_slh_sign PRIVATE "${DNA_ROOT}/shared")
target_link_libraries(ckb_slh_sign PRIVATE qgp_slhdsa)
set_target_properties(ckb_slh_sign PROPERTIES OUTPUT_NAME ckb-slh-sign SUFFIX .wasm)
target_link_options(ckb_slh_sign PRIVATE -O2 --no-entry "-sSTANDALONE_WASM=1" "-sFILESYSTEM=0" "-sSTACK_SIZE=1048576" "-sINITIAL_MEMORY=4194304" "-sALLOW_MEMORY_GROWTH=0" "-sEXPORTED_FUNCTIONS=['_ckb_slh_seed','_ckb_slh_message','_ckb_slh_public_key','_ckb_slh_signature','_ckb_slh_sign','_ckb_slh_clear','_ckb_slh_random','_ckb_slh_verify']" -Wl,--export-memory -Wl,--gc-sections -Wl,--strip-all)
